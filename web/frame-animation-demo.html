const player = new FrameAnimationPlayer();
await player.load({
  name: 'shockwave',
  imageUrl: 'assets/anims/shockwave.png',
  frameWidth: 128, frameHeight: 128,
  cols: 8, rows: 1, count: 8,
  fps: 24, loop: true,
  onComplete: () => console.log('done')
});

function tick(dt) {
  player.update(dt);
  player.draw(ctx, 300, 200, {scale:2, alpha:.9});
}
// 替换前：Spine 方案
// 1. 美术出 3 组 .skel+.atlas+.png（3×3 = 9 个文件）
// 2. 集成 spine-3.8 运行时（~300 KB）
// 3. 手写 setAnimation + trackEntry 监听
// 4. 骨骼变换矩阵 / 蒙皮 mesh 每帧计算

const spine = new Spine(skelData, atlas);
spine.state.setAnimation(0, 'attack', false);
spine.skeleton.x = x; spine.skeleton.y = y;
spine.update(dt); spine.draw(ctx); /* 每帧 CPU 重算所有骨骼 */

// 替换后：序列帧方案
// 1. 美术导出 3 张 sprite sheet（3 个 PNG）
// 2. JSON 描述帧数/尺寸/帧率（<20 行）
// 3. 无运行时，Player <10 KB（本页提供）
// 4. 纯 drawImage 切图：GPU 友好，CPU 零开销

const fa = new FrameAnimationPlayer();
fa.loadSync(ANIM.shockwave);      // 内联 JSON
fa.play('attack');
fa.update(dt);
fa.draw(ctx, x, y, {scale:2}); /* drawImage 坐标直接切 */

// 制作成本对比（以 3 个特效为例）
// Spine: 美术 rig + 权重 + K帧 + 导出调试 ≈ 6-8 小时
// 帧动画: 序列输出 + 打图集 + 写 JSON   ≈ 30-60 分钟
// 结论：序列帧可覆盖 80% 的 2D 特效需求。
